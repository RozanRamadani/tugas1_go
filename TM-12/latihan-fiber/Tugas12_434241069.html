// D.2.1: Swap ke kelas yang kuotanya penuh harus ditolak 409 COURSE_FULL,
// mahasiswa tetap di kelas asal, enrolled_count tidak berubah, dan outbox kosong.
func TestEnrollEndpoint_SwapCourseFull(t *testing.T) {
	a := newAPI(t)
	student := seedStudent(t)
	targetStudent := seedStudent(t)

	courseA := seedCourse(t, 3, 40) // kelas asal
	courseB := seedCourse(t, 3, 1)  // kelas tujuan dengan kuota 1

	if status, _ := a.enroll(t, student, courseA.ID); status != http.StatusCreated {
		t.Fatalf("enroll kelas A: status = %d, seharusnya 201", status)
	}
	if status, _ := a.enroll(t, targetStudent, courseB.ID); status != http.StatusCreated {
		t.Fatalf("enroll target student kelas B: status = %d, seharusnya 201", status)
	}

	token, _ := a.jwt.GenerateAccessToken(model.AuthUser{UserID: student.ID, Username: student.Username, Role: "user"})
	body := `{"from_course_id":` + strconv.Itoa(courseA.ID) + `,"to_course_id":` + strconv.Itoa(courseB.ID) + `}`
	req := httptest.NewRequest(http.MethodPost, "/api/v1/enrollments/swap", strings.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+token)
	req.Header.Set("Content-Type", "application/json")

	resp, err := a.app.Test(req, -1)
	if err != nil {
		t.Fatalf("request swap: %v", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusConflict {
		t.Errorf("status swap = %d, seharusnya 409 Conflict", resp.StatusCode)
	}

	enrolledA := countRows(t, `SELECT COUNT(*) FROM enrollments WHERE user_id = $1 AND course_id = $2`, student.ID, courseA.ID)
	if enrolledA != 1 {
		t.Errorf("mahasiswa harus tetap terdaftar di kelas asal (A)")
	}

	cA := countRows(t, `SELECT enrolled_count FROM courses WHERE id = $1`, courseA.ID)
	cB := countRows(t, `SELECT enrolled_count FROM courses WHERE id = $1`, courseB.ID)
	if cA != 1 || cB != 1 {
		t.Errorf("enrolled_count berubah: cA=%d, cB=%d, seharusnya tetap 1", cA, cB)
	}

	swappedEvents := countRows(t, `SELECT COUNT(*) FROM outbox WHERE (payload->>'user_id')::INT = $1 AND event_type = $2`,
		student.ID, model.EventEnrollmentSwapped)
	if swappedEvents != 0 {
		t.Errorf("event enrollment.swapped = %d, seharusnya 0", swappedEvents)
	}
}

// D.2.2: Drop kelas berhasil menghasilkan 204 No Content, baris enrollment hilang,
// enrolled_count berkurang satu, dan outbox mencatat event enrollment.dropped.
func TestEnrollEndpoint_DropSuccess(t *testing.T) {
	a := newAPI(t)
	student := seedStudent(t)
	course := seedCourse(t, 3, 40)

	if status, _ := a.enroll(t, student, course.ID); status != http.StatusCreated {
		t.Fatalf("enroll gagal: status = %d", status)
	}

	token, _ := a.jwt.GenerateAccessToken(model.AuthUser{UserID: student.ID, Username: student.Username, Role: "user"})
	req := httptest.NewRequest(http.MethodDelete, "/api/v1/courses/"+strconv.Itoa(course.ID)+"/enroll", nil)
	req.Header.Set("Authorization", "Bearer "+token)
	req.Header.Set("Content-Type", "application/json")

	resp, err := a.app.Test(req, -1)
	if err != nil {
		t.Fatalf("request drop: %v", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusNoContent {
		t.Fatalf("status drop = %d, seharusnya 204", resp.StatusCode)
	}

	enrolled := countRows(t, `SELECT COUNT(*) FROM enrollments WHERE user_id = $1 AND course_id = $2`, student.ID, course.ID)
	if enrolled != 0 {
		t.Errorf("baris enrollment = %d, seharusnya 0 setelah drop", enrolled)
	}

	count := countRows(t, `SELECT enrolled_count FROM courses WHERE id = $1`, course.ID)
	if count != 0 {
		t.Errorf("enrolled_count = %d, seharusnya 0 setelah drop", count)
	}

	dropEvents := countRows(t, `SELECT COUNT(*) FROM outbox WHERE (payload->>'user_id')::INT = $1 AND event_type = $2`,
		student.ID, model.EventEnrollmentDropped)
	if dropEvents != 1 {
		t.Errorf("event enrollment.dropped = %d, seharusnya tepat 1", dropEvents)
	}
}

D.4 Analisis Singkat

1. Coverage vs. Mutation Testing pada Kasus TestWithinCreditLimit

Mengapa coverage tidak dapat menggantikan uji mutation?
Coverage (khususnya statement/line coverage) hanya mencatat apakah sebuah baris kode pernah dieksekusi saat test suite berjalan, namun sama sekali tidak menilai apakah logika, batas kondisi tepi, dan hasil eksekusi tersebut diverifikasi dengan teliti melalui assertion. Sebagai contoh nyata, fungsi WithinCreditLimit(taken, adding int) bool { return taken + adding <= 24 } hanya terdiri dari satu baris logika. Jika test hanya menyertakan skenario taken: 0, adding: 3, coverage fungsi tersebut langsung tercatat 100.0%. Namun, apabila operator diubah menjadi cacat logika (< alih-alih <=), test tersebut tetap lulus (PASS). Uji mutasi membuktikan bahwa coverage 100% dapat memberikan rasa aman palsu jika tidak diuji dengan mutasi kode produksi.

Mengapa uji mutation tidak dapat menggantikan coverage?
Uji mutasi hanya dapat diterapkan secara efektif pada kode yang sudah memiliki test awal. Jika suatu bagian kode atau percabangan sama sekali belum pernah dieksekusi oleh test (coverage 0%), mutasi pada bagian tersebut tidak ada gunanya karena pasti tidak ada test yang mengujinya. Selain itu, uji mutasi memerlukan biaya komputasi yang sangat tinggi (harus mengompilasi dan mengeksekusi test berulang kali untuk setiap mutasi sintaks). Coverage memberikan peta awal yang cepat untuk mengidentifikasi bagian kode mana yang belum tersentuh (blind spot), sedangkan mutation testing menguji ketajaman kualitas pengujian pada area yang telah tersentuh. Keduanya bersifat komplementer.

2. Keterbatasan Fake Repository Dibandingkan MongoDB Sungguhan

Perilaku MongoDB sungguhan yang TIDAK ditiru fake:
Fake repository pada Langkah 2 meniru unique index dengan perulangan linier di slice memory Go yang dilindungi oleh sync.Mutex lokal. Perilaku MongoDB sungguhan yang tidak ditiru adalah penegakan Distributed Unique Index di level storage engine WiredTiger saat menerima konkurensi request tinggi. Pada MongoDB sungguhan, duplikasi index memicu error driver spesifik (kode error 11000 duplicate key error) dengan detail key pattern tertentu, serta memiliki karakteristik latensi I/O jaringan, transaksi multi-dokumen, partial filter expressions, dan perilaku connection pooling.

Jenis test yang seharusnya memeriksanya:
Integration Test terhadap instans MongoDB sungguhan (atau database test container). Integration test menghubungkan repository ke instance MongoDB riil, memastikan indeks unik dibuat dengan benar (CreateOne(ctx, IndexModel{...})), dan menguji bahwa operasi insert konkuren yang bertabrakan benar-benar menghasilkan error driver 11000 yang berhasil diterjemahkan oleh repository menjadi repository.ErrDuplicate.

3. Perbandingan Mekanisme Migrasi Docker vs. In-App Migrator (Multi-Replika)

Perbandingan:
Pada Docker Compose bawaan modul, migrasi dijalankan melalui volume binding /docker-entrypoint-initdb.d yang hanya dieksekusi satu kali saat volume PostgreSQL masih kosong. Sebaliknya, in-app migration (seperti pada TestMain) mengeksekusi DDL migration secara terprogram setiap kali aplikasi menyala.

Kelebihan dan Risiko jika API dijalankan 3 replika sekaligus:
• Kelebihan: Otomatisasi deployment sangat tinggi. Setiap container baru langsung memastikan skema database mutakhir tanpa memerlukan intervensi manual dari administrator database.
• Risiko Kritis: Race Condition DDL dan Deadlock. Ketika ketiga replika API menyala bersamaan, ketiganya akan berusaha mengeksekusi script DDL migrasi (CREATE TABLE, ALTER TABLE, CREATE INDEX) ke database PostgreSQL yang sama secara paralel. Tanpa mekanisme Distributed Lock (seperti pg_advisory_lock), migrasi dapat gagal di tengah jalan dengan error tuple concurrently updated atau skema corrupt. Solusi industri terbaik adalah menjalankan migrasi sebagai init-container / migration job terpisah sebelum replika API dinyalakan.

4. Risiko Keamanan Swagger UI Terbuka di Server Kampus & Port 8080

Dua Risiko Konkret:
1. Information Disclosure & Attack Surface Reconnaissance: Swagger UI mengekspos rincian seluruh endpoint privat (seperti /admin/queue, /users/:id/role), struktur payload JSON, tipe parameter, dan format token autentikasi. Penyerang di internet tidak perlu melakukan blind probing; mereka mendapatkan blueprint arsitektur backend secara cuma-cuma untuk merancang eksploitasi payload yang presisi.
2. Brute Force & Unrestricted API Abuse: Port 8080 diekspos langsung ke internet tanpa melewati Web Application Firewall (WAF) atau reverse proxy. Endpoint /auth/login dan /auth/register rentan diserang credential stuffing atau serangan Denial of Service (DoS) yang dapat menguras resource container backend dan koneksi database pool.

Perbaikan yang Tepat:
1. Nonaktifkan Swagger UI pada environment production/public dengan menyetel SWAGGER_ENABLED=false pada file environment server. Jika dokumentasi tetap dibutuhkan oleh pengembang, pasang autentikasi lapis dua (HTTP Basic Auth) atau batasi akses Swagger hanya untuk subnet IP jaringan internal kampus (VPN Kampus).
2. Tempatkan Reverse Proxy (seperti Nginx, Caddy, atau Cloudflare WAF) di depan port aplikasi untuk menerapkan enkripsi TLS/HTTPS (port 443), rate limiting ketat pada endpoint otentikasi, dan memblokir akses publik ke path /admin/* serta /swagger/*.
